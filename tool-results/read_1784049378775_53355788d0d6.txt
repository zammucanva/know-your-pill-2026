     8→/**
     9→ * Drug class identifiers.
    10→ *
    11→ * The first 8 (`depressant`..`ssri`) describe substances of abuse and
    12→ * major medication classes — they have stable accent colors defined in
    13→ * `classes.ts` and CSS variables in `globals.css`.
    14→ *
    15→ * The remaining IDs cover other medication classes that don't need a
    16→ * distinct accent color (they fall back to the brand teal).
    17→ */
    18→export type DrugClassId =
    19→  // Substances of abuse (with accent colors)
    20→  | "depressant"
    21→  | "stimulant"
    22→  | "hallucinogen"
    23→  | "opioid"
    24→  | "cannabinoid"
    25→  | "dissociative"
    26→  | "inhalant"
    27→  // Medication classes
    28→  | "ssri"
    29→  | "snri"
    30→  | "tca"
    31→  | "maoi"
    32→  | "atypical-antipsychotic"
    33→  | "typical-antipsychotic"
    34→  | "mood-stabiliser"
    35→  | "benzodiazepine"
    36→  | "non-benzodiazepine-hypnotic";
